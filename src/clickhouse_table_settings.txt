# ClickHouse settings pool used by pstress (--table-settings-file).
#
# Line format:
#
#   [session:][<prob>:]<name> = <values>
#
#   <prob>    0-100, the chance this setting is applied. Omitted means 100.
#             Table lines are rolled independently for every table, so a run
#             with 20 tables gets 20 different SETTINGS clauses.
#             Session lines are rolled once per run.
#   <values>  v1|v2|v3   -> one value picked at random
#             int:lo..hi -> a random integer in [lo, hi]
#   session:  applied as "SET <name> = <value>" on every connection, before any
#             DDL runs. Needed for the allow_experimental_* flags that gate
#             experimental table settings — without the session flag the server
#             rejects the CREATE TABLE outright.
#
# Blank lines and lines starting with # are ignored.
#
# WARNING: only add table settings that cannot change query RESULTS. pstress
# verifies row counts and checksums across replicas, so a setting that changes
# what a SELECT returns shows up as a false-positive data mismatch. In
# particular do NOT add deduplication windows, TTL settings, or
# parts_to_throw_insert / parts_to_delay_insert.
#
# enable_block_number_column and enable_block_offset_column are always emitted
# by pstress (the lightweight mutation paths depend on them) and cannot be
# overridden from this file.

# ---- part format: compact vs wide ----
 50:min_bytes_for_wide_part                        = 0|1048576|10485760
 50:min_rows_for_wide_part                         = 0|1000|100000

# ---- index granularity ----
100:index_granularity                              = 1024|4096|8192|16384
 30:index_granularity_bytes                        = 1048576|10485760

# ---- merge behaviour ----
 30:merge_max_block_size                           = 1024|8192|65536
 50:enable_vertical_merge_algorithm                = 0|1
 20:vertical_merge_algorithm_min_rows_to_activate  = int:1..1000
 10:min_age_to_force_merge_seconds                 = 0|10
 10:old_parts_lifetime                             = 10|480

# ---- compression / serialization ----
 20:min_compress_block_size                        = 8192|65536
 20:max_compress_block_size                        = 65536|1048576
 25:compress_marks                                 = 0|1
 25:compress_primary_key                           = 0|1
 10:marks_compress_block_size                      = 16384|65536
 10:primary_key_compress_block_size                = 16384|65536
 25:ratio_of_defaults_for_sparse_serialization     = 0.9|1.0

# fill in the names. prob 100 means an unknown setting name aborts the run
# rather than silently leaving the feature untested.
#
# session:allow_experimental_<feature> = 1
# 100:<feature_setting> = 1

# https://github.com/ClickHouse/ClickHouse/pull/111834 — merges and mutations

# session: line: this is an EXPERIMENTAL-*tier* MergeTree setting, gated only by
# the server-config allow_feature_tier, not by an allow_experimental_* session
# flag. Merge/mutation only, so the run needs --optimize and --ch-alter-update

100:allow_experimental_adaptive_codec_selection     = 1
